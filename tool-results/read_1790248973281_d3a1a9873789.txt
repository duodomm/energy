   321→function LoadBuilderSlot({
   322→  appliances, setAppliances, input, set,
   323→}: {
   324→  appliances: NonNullable<CalcInput["appliances"]>
   325→  setAppliances: (a: CalcInput["appliances"]) => void
   326→  input: CalcInput
   327→  set: (patch: Partial<CalcInput>) => void
   328→}) {
   329→  // Синхронизация конструктора с полями расчётного ядра
   330→  useEffect(() => {
   331→    const s = summarize(appliances, input.startK)
   332→    if (Math.abs(s.dailyKwh - input.dailyKwh) > 0.05 || Math.abs(s.peakKw - input.peakKw) > 0.05) {
   333→      set({ dailyKwh: s.dailyKwh, peakKw: s.peakKw })
   334→    }
   335→  }, [appliances, input.startK])
   336→
   337→  return (
   338→    <div className="mb-6">
   339→      <LoadBuilder objectType={input.objectType} appliances={appliances} onChange={setAppliances} startK={input.startK} />
   340→    </div>
   341→  )
   342→}
